cmake_minimum_required(VERSION 3.8)
project(anc_nav2_sequential_controller)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_lifecycle rcl_interfaces nav2_core nav2_util nav2_costmap_2d tf2 tf2_ros tf2_geometry_msgs geometry_msgs nav_msgs std_msgs pluginlib)
  find_package(${dep} REQUIRED)
endforeach()
add_library(anc_sequential_controller SHARED src/sequential_controller.cpp)
target_include_directories(anc_sequential_controller PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(anc_sequential_controller PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(anc_sequential_controller rclcpp rclcpp_lifecycle rcl_interfaces nav2_core nav2_util nav2_costmap_2d tf2 tf2_ros tf2_geometry_msgs geometry_msgs nav_msgs std_msgs pluginlib)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS anc_sequential_controller ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(primitive_tests test/primitive_tests.cpp)
  target_include_directories(primitive_tests PRIVATE include)
  find_package(laser_geometry REQUIRED)
  ament_add_gtest(costmap_tests test/costmap_tests.cpp)
  ament_target_dependencies(costmap_tests nav2_costmap_2d geometry_msgs laser_geometry)
endif()
ament_export_include_directories(include)
ament_export_libraries(anc_sequential_controller)
ament_package()
